# Define the rate-limiting zone
limit_req_zone $binary_remote_addr zone=bad_requests:20m rate=2r/s;

server {
    # Trust Cloudflare IPs and read the real IP from the header
    set_real_ip_from 0.0.0.0/0;
    real_ip_header CF-Connecting-IP;

    if ($host = www.lighthouse-tutors.com) {
        return 301 https://$host$request_uri;
    }

    if ($host = lighthouse-tutors.com) {
        return 301 https://$host$request_uri;
    }

    return 302 https://$server_name$request_uri;

    listen 80;
    server_name lighthouse-tutors.com www.lighthouse-tutors.com;

    location / {
        include proxy_params;
        proxy_pass http://unix:/run/gunicorn.sock;
        limit_req zone=bad_requests burst=5 nodelay;
        limit_req_status 429;
        client_max_body_size 10M;
    }

    # Apply the rate-limiting for other relevant locations
    location /static/ {
        alias /home/django/django_edu/tutors_django/staticfiles;
        limit_req zone=bad_requests burst=5 nodelay;
    }

    location /media/ {
        alias /home/django/django_edu/tutors_django/media;
        limit_req zone=bad_requests burst=5 nodelay;
    }

    # WebSocket proxying
    location /ws/chess/ {
        proxy_pass http://127.0.0.1:8001/ws/chess/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_redirect off;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Host $server_name;
        add_header Access-Control-Allow-Origin https://www.lighthouse-tutors.com;
    }

    location ~ /\.env$ {
        deny all;
        access_log off;
        log_not_found off;
    }

    location ~ /\.git {
        deny all;
        access_log off;
        log_not_found off;
    }

    location ~* "\.php" {
        deny all;
        return 403;
    }
}
