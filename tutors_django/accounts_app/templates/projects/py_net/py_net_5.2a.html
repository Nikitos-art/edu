This custom class verifies that if a user entered a host address rather than a network address, converts
the host address to a network address and prints the network address and the mask. 

        from masks import subnet_masks

        MARKER = '-----------------------------------'


        class IpNetHostMask:
            def __init__(self, net_addr, masks=subnet_masks):
                self.net_addr = net_addr.split("/")
                self.masks = masks
                self.network = self.net_addr[0].split(".")
                self.ip_mask = int(self.net_addr[1])

            def process(self):
                net_bin = self.net_addr_to_bin()
                network_result = self.validator()
                mask_result = f"{self.ip_mask}\n{subnet_masks[self.ip_mask][0]}\n{subnet_masks[self.ip_mask][1]}"
                return f"{network_result}\n{net_bin} \n{MARKER}\n{mask_result}\n{MARKER}\n"

            def net_addr_to_bin(self):
                result = ""
                for byte in self.network:
                    byte_elem = bin(int(byte))[2:]
                    padding_length = (8 - len(byte_elem) % 8) % 8
                    binary_representation = f"{'0' * padding_length + byte_elem} "
                    result += binary_representation
                return result

            def validator(self):
                network_to_bin_no_whitespaces_final = ""
                bin_net = self.net_addr_to_bin().replace(" ", "")
                print(MARKER, bin_net, MARKER)
                for ind, bi in enumerate(bin_net):
                    if bi == " ":
                        pass
                    elif ind <= self.ip_mask:
                        network_to_bin_no_whitespaces_final += bi
                    else:
                        network_to_bin_no_whitespaces_final += '0'

                octets = [network_to_bin_no_whitespaces_final[i:i + 8] for i in range(0, 32, 8)]
                decimal_octets = [str(int(octet, 2)) for octet in octets]
                decimal_ip = '.'.join(decimal_octets)

                return f"{decimal_ip}"

            def __str__(self):
                return f"{self.process()}"
Explanation:
   Here I first import all the common used subnet masks (32 in total) with their corresponding ip ranges/values. The main method
of this class is the process(). That's where network address is converted to binary, and the validator() method is called, which
checks that the address is indeed a host and not a network address, and otherwise makes adjustments as described in the exercise's
task. A dunder __str__ method is added for nice string representation.
